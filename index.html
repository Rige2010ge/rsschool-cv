My solution: 
        let nodeArray = [];
        
        function loop_size(node) {
            let nextNode = node.getNext();
            let collection = new Set();
        
            while (true) {
                if (node !== nextNode) {
                    let hasNode = collection.has(node);
                    collection.add(node);
                    if (hasNode) {
                        let index;
                        [...collection].forEach((el, indexArr) => {
                        if (el === node) index = indexArr;
                        });
                        return collection.size - index;
                    } else {
                        node = nextNode;
                        nextNode = node.getNext();
                    }
                } else {
                    return 1;
                }
            }
        }
